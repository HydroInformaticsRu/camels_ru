%% Figure 2 (fig:workflow) — CAMELS-RU production workflow schematic.
%% Standalone TikZ source; sans-serif labels match the manuscript plotting style.
%% Build (from repo root):
%%   pdflatex -output-directory=paper/figures_src paper/figures_src/fig_workflow.tex

\documentclass[border=2mm]{standalone}
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{tikz}
\usetikzlibrary{positioning,arrows.meta}

\begin{document}
\begin{tikzpicture}[
    font=\small,
    box/.style={draw, rounded corners=2pt, align=center, minimum height=12mm, inner sep=2mm, line width=0.45pt},
    src/.style={box, fill=black!3, draw=black!35, text width=36mm},
    proc/.style={box, fill=blue!4, draw=black!45, text width=46mm},
    outfile/.style={box, fill=black!3, draw=black!35, text width=36mm},
    flow/.style={-{Stealth[length=2mm]}, line width=0.65pt, draw=black!60},
  ]

  % Column x-positions and row y-positions (row-aligned: horizontal arrows only)
  \def\xsrc{0}    \def\xproc{5.5}  \def\xout{11}

  % Row 1 — discharge and stage
  \node[src] (s1) at (\xsrc,0) {AIS GMVO portal\\ daily discharge and stage\\ 2008--2023};
  \node[proc] (p1) at (\xproc,0) {Parsing and harmonisation\\ gap filling $\leq 6$\,d\\ reservoir stage: $\leq 15$\,d};
  \node[outfile]  (o1) at (\xout,0)  {\textsf{discharge.nc}\\ \textsf{water\_level.nc}};

  % Row 2 — quality grading (fed by row 1)
  \node[proc] (p2) at (\xproc,-1.9) {Discharge-only quality grading\\ completeness + 16 flags\\ $\rightarrow$ grades A--F};
  \node[outfile]  (o2) at (\xout,-1.9)  {\textsf{year\_grades.csv}\\ \textsf{year\_flags.csv}};

  % Row 3 — delineation
  \node[src, font=\footnotesize] (s3) at (\xsrc,-3.8) {MERIT Hydro DEM\\ AIS GMVO + Hydroweb:\\ gauge coordinates\\ Hydroweb: reference areas};
  \node[proc] (p3) at (\xproc,-3.8) {Watershed delineation,\\ manual boundary verification};
  \node[outfile]  (o3) at (\xout,-3.8)  {\textsf{boundaries.gpkg}};

  % Row 4 — boundaries define the basin aggregation domain
  \node[src]  (s4) at (\xsrc,-5.7)  {Gridded forcing\\ MSWEP, ERA5-Land,\\ GPCP, GLEAM4\\ + \textsf{boundaries.gpkg}};
  \node[proc] (p4) at (\xproc,-5.7) {Basin-averaged\\ forcing aggregation};
  \node[outfile]  (o4) at (\xout,-5.7)  {\textsf{forcing.nc}};

  % Row 5 — boundaries also define attribute extraction
  \node[src]  (s5) at (\xsrc,-7.6)  {HydroATLAS v1.0\\ + \textsf{boundaries.gpkg}};
  \node[proc] (p5) at (\xproc,-7.6) {Attribute extraction};
  \node[outfile]  (o5) at (\xout,-7.6)  {\textsf{attributes.csv}};

  % Row 6 — all three signature inputs are explicit
  \node[src]  (s6) at (\xsrc,-9.5)  {Discharge, forcing,\\ catchment area};
  \node[proc] (p6) at (\xproc,-9.5) {Signature computation};
  \node[outfile]  (o6) at (\xout,-9.5)  {\textsf{signatures.csv}};

  % Column headers
  \node[font=\normalsize\bfseries] at (\xsrc,1.15)  {Sources};
  \node[font=\normalsize\bfseries] at (\xproc,1.15) {Processing and screening};
  \node[font=\normalsize\bfseries] at (\xout,1.15)  {Released files};

  % Horizontal flows (row-aligned, no crossings)
  \foreach \r in {1,3,4,5,6} \draw[flow] (s\r) -- (p\r);
  \foreach \r in {1,2,3,4,5,6} \draw[flow] (p\r) -- (o\r);

  % Vertical flows inside the processing column
  \draw[flow] (p1) -- (p2) node[midway, right=1mm, font=\footnotesize\itshape] {discharge};

\end{tikzpicture}
\end{document}
